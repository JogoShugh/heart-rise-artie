Kept from the HAL-FORMS walk (hal-forms/release.heart.yaml)
steps:
  - act: scm:pulls-get
    fields:
      owner: ${owner}
      repo: ${repo}
      pull_number: ${pull}
    intent: Open pull request ${pull} of ${owner}/${repo}
    then:
      - keep-as: pr
  - enter: ${watcher}
    steps:
      - ask: watcher:watch
        fields:
          resource: ${pr.@self}
          offers: scm:merge
        reply: watch
  - once-answered:
      reply: watch
      steps:
        - enter: ${pr.@self}
          steps:
            - act: scm:merge
              fields:
                method: squash
              intent: merge it by squashing
              then:
                - check:
                    - field: mergeable_state
                      is: unknown
                    - field: merged
                      is: true
                    - field: state
                      is: closed
                - keep-as: merged
  - enter: ${jenkins}
    steps:
      - ask: ci:build
        fields:
          commit: ${merged.merge_commit_sha}
        reply: build
  - once-answered:
      reply: build
      steps:
        - follow: hr:in-reply-to
          intent: follow its in-reply-to link to the build
          then:
            - run-script: ci:smoke-test
              on-link: ci:deployment
              inputs:
                book: ${book}
                author: ${author}
                renewable: ${renewable}
                problem: ${problem}
                comment: ${comment}
              as: smoke
  - given:
      - at: smoke
        field: passed
        is: true
    then:
      - enter: ${dev}
        steps:
          - act: env:get-environment
            fields:
              id: ${environment}
            inferred: open environment ${environment}
            from-intent: open environment ${environment} and promote commit {merged.merge_commit_sha}, citing {smoke.passing} of {smoke.of} scenarios passed
            then:
              - for-each: environment
                then:
                  - when-offered:
                      - act: env:promote
                        fields:
                          commit: ${merged.merge_commit_sha}
                          smoke: # review: the reasoner chose it, asked "promote commit 39590c7cd6242a278c59c230eb17f387161f38be, with the smoke field as an object: {"passed": 5, "of": 5} (5 of 5 scenarios passed)"
                            passed: ${smoke.passing}
                            of: ${smoke.of}
                        intent: open environment ${environment} and promote commit {merged.merge_commit_sha}, citing {smoke.passing} of {smoke.of} scenarios passed
                        then:
                          - check:
                              - field: commit
                                is: ${merged.merge_commit_sha}
                    intent: open environment ${environment} and promote commit {merged.merge_commit_sha}, citing {smoke.passing} of {smoke.of} scenarios passed
                    otherwise:
                      - note: none of these acts is offered here
    otherwise:
      - at: ${pr.@self}
        steps:
          - act: scm:comment
            fields:
              body: "Not promoted to Dev: smoke scenarios ${smoke.failed} failed."
