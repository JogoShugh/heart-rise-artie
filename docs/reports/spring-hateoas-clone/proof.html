package org.springframework.hateoas.examples;

import static org.springframework.hateoas.server.mvc.WebMvcLinkBuilder.*;

import java.net.URI;
import java.net.URISyntaxException;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.StreamSupport;

import org.springframework.hateoas.CollectionModel;
import org.springframework.hateoas.EntityModel;
import org.springframework.hateoas.IanaLinkRelations;
import org.springframework.hateoas.Link;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

/**
 * @author Greg Turnquist
 */
@RestController
class EmployeeController {

	private final EmployeeRepository repository;

	EmployeeController(EmployeeRepository repository) {
		this.repository = repository;
	}

	@GetMapping("/employees")
	ResponseEntity<CollectionModel<EntityModel<Employee>>> findAll() {

		List<EntityModel<Employee>> employeeResources = StreamSupport.stream(repository.findAll().spliterator(), false)
				.map(employee -> EntityModel.of(employee,
						linkTo(methodOn(EmployeeController.class).findOne(employee.getId())).withSelfRel()
								.andAffordance(afford(methodOn(EmployeeController.class).updateEmployee(null, employee.getId())))
								.andAffordance(afford(methodOn(EmployeeController.class).deleteEmployee(employee.getId()))),
						linkTo(methodOn(EmployeeController.class).findAll()).withRel("employees")))
				.collect(Collectors.toList());

		return ResponseEntity.ok(CollectionModel.of( //
				employeeResources, //
				linkTo(methodOn(EmployeeController.class).findAll()).withSelfRel()
						.andAffordance(afford(methodOn(EmployeeController.class).newEmployee(null)))));
	}

	@PostMapping("/employees")
	ResponseEntity<?> newEmployee(@RequestBody Employee employee) {

		Employee savedEmployee = repository.save(employee);

		return EntityModel.of(savedEmployee,
				linkTo(methodOn(EmployeeController.class).findOne(savedEmployee.getId())).withSelfRel()
						.andAffordance(afford(methodOn(EmployeeController.class).updateEmployee(null, savedEmployee.getId())))
						.andAffordance(afford(methodOn(EmployeeController.class).deleteEmployee(savedEmployee.getId()))),
				linkTo(methodOn(EmployeeController.class).findAll()).withRel("employees")).getLink(IanaLinkRelations.SELF)
						.map(Link::getHref) //
						.map(href -> {
							try {
								return new URI(href);
							} catch (URISyntaxException e) {
								throw new RuntimeException(e);
							}
						}) //
						.map(uri -> ResponseEntity.noContent().location(uri).build())
						.orElse(ResponseEntity.badRequest().body("Unable to create " + employee));
	}

	@GetMapping("/employees/{id}")
	ResponseEntity<EntityModel<Employee>> findOne(@PathVariable long id) {

		return repository.findById(id)
				.map(employee -> EntityModel.of(employee,
						linkTo(methodOn(EmployeeController.class).findOne(employee.getId())).withSelfRel()
								.andAffordance(afford(methodOn(EmployeeController.class).updateEmployee(null, employee.getId())))
								.andAffordance(afford(methodOn(EmployeeController.class).deleteEmployee(employee.getId()))),
						linkTo(methodOn(EmployeeController.class).findAll()).withRel("employees")))
				.map(ResponseEntity::ok) //
				.orElse(ResponseEntity.notFound().build());
	}

	@PutMapping("/employees/{id}")
	ResponseEntity<?> updateEmployee(@RequestBody Employee employee, @PathVariable long id) {

		Employee employeeToUpdate = employee;
		employeeToUpdate.setId(id);

		Employee updatedEmployee = repository.save(employeeToUpdate);

		return EntityModel.of(updatedEmployee,
				linkTo(methodOn(EmployeeController.class).findOne(updatedEmployee.getId())).withSelfRel()
						.andAffordance(afford(methodOn(EmployeeController.class).updateEmployee(null, updatedEmployee.getId())))
						.andAffordance(afford(methodOn(EmployeeController.class).deleteEmployee(updatedEmployee.getId()))),
				linkTo(methodOn(EmployeeController.class).findAll()).withRel("employees")).getLink(IanaLinkRelations.SELF)
						.map(Link::getHref).map(href -> {
							try {
								return new URI(href);
							} catch (URISyntaxException e) {
								throw new RuntimeException(e);
							}
						}) //
						.map(uri -> ResponseEntity.noContent().location(uri).build()) //
						.orElse(ResponseEntity.badRequest().body("Unable to update " + employeeToUpdate));
	}

	@DeleteMapping("/employees/{id}")
	ResponseEntity<?> deleteEmployee(@PathVariable long id) {

		repository.deleteById(id);

		return ResponseEntity.noContent().build();
	}
}

StockSpring.kt, the clone
package com.artie.heart.sims.employees

import com.fasterxml.jackson.databind.ObjectMapper
import com.fasterxml.jackson.databind.node.ObjectNode
import org.http4k.core.HttpHandler
import org.http4k.core.Method.DELETE
import org.http4k.core.Method.GET
import org.http4k.core.Method.POST
import org.http4k.core.Method.PUT
import org.http4k.core.Request
import org.http4k.core.Response
import org.http4k.core.Status
import org.http4k.routing.bind
import org.http4k.routing.path
import org.http4k.routing.routes

/**
 * The employees as Spring HATEOAS's own affordances example
 * (spring-projects/spring-hateoas-examples, module affordances) serves them out of the box,
 * byte for byte, copied from a run of it on 1 November 2026 and checked against it since
 * (SpringCloneTrace). It is no HEART service, and does nothing a HEART service would:
 * HAL-FORMS only, its templates' methods in lower case, no profile, no ETag, absolute links,
 * and every act answered 204 with no body (a create and an update name the employee in
 * Location). Asked for application/hal+json, it answers its own JSON: a `links`
 * array, no forms.
 */
class StockSpring(private val staff: Staff) {
    private val mapper = ObjectMapper()

    val handler: HttpHandler = routes(
        "/employees" bind GET to { request -> list(request) },
        "/employees" bind POST to { request ->
            val made = fromBody(request, null)
            Response(Status.NO_CONTENT).header("Location", "${origin(request)}/employees/${made.id}")
        },
        "/employees/{id}" bind GET to { request ->
            staff.get(idOf(request))?.let { one(request, it) } ?: Response(Status.NOT_FOUND)
        },
        "/employees/{id}" bind PUT to { request ->
            val saved = fromBody(request, idOf(request))
            Response(Status.NO_CONTENT).header("Location", "${origin(request)}/employees/${saved.id}")
        },
        "/employees/{id}" bind DELETE to { request ->
            staff.remove(idOf(request))
            Response(Status.NO_CONTENT)
        },
    )

    /** Spring's save: a new employee when there is no [id], the employee [id] as given otherwise. */
    private fun fromBody(request: Request, id: Long?): Employee {
        val body = mapper.readTree(request.bodyString())
        fun text(name: String) = body.path(name).asText("")
        return if (id == null) {
            staff.hire(text("firstName"), text("lastName"), text("role"))
        } else {
            staff.save(id, text("firstName"), text("lastName"), text("role"))
        }
    }

    private fun idOf(request: Request) = request.path("id")?.toLongOrNull() ?: -1

    /** Where the request came: Spring writes every link absolute, against the address it was asked at. */
    private fun origin(request: Request) =
        "http://${request.header(HOST_HEADER) ?: request.header("Host") ?: request.uri.authority}"

    private fun halForms(request: Request) = request.header("Accept").orEmpty().let { HAL_FORMS in it || HAL !in it }

    private fun list(request: Request): Response {
        val origin = origin(request)
        val body = if (halForms(request)) {
            mapper.createObjectNode().apply {
                putObject("_embedded").putArray("employees").apply { staff.working().forEach { add(employee(it, origin)) } }
                putObject("_links").putObject("self").put("href", "$origin/employees")
                set<ObjectNode>("_templates", templates(mapOf("default" to "post")))
            }
        } else {
            mapper.createObjectNode().apply {
                putArray("links").addObject().put("rel", "self").put("href", "$origin/employees")
                putArray("content").apply { staff.working().forEach { add(plain(it, origin)) } }
            }
        }
        return answer(request, body)
    }

    private fun one(request: Request, employee: Employee): Response {
        val origin = origin(request)
        return answer(request, if (halForms(request)) employee(employee, origin) else plain(employee, origin))
    }

    private fun answer(request: Request, body: ObjectNode) = Response(Status.OK)
        .header("Content-Type", if (halForms(request)) HAL_FORMS else HAL)
        .body(mapper.writeValueAsString(body))

    private fun fields(employee: Employee): ObjectNode = mapper.createObjectNode()
        .put("id", employee.id).put("firstName", employee.firstName)
        .put("lastName", employee.lastName).put("role", employee.role)

    private fun employee(employee: Employee, origin: String): ObjectNode = fields(employee).apply {
        putObject("_links").apply {
            putObject("self").put("href", "$origin/employees/${employee.id}")
            putObject("employees").put("href", "$origin/employees")
        }
        set<ObjectNode>("_templates", templates(mapOf("default" to "put", "deleteEmployee" to "delete")))
    }

    private fun plain(employee: Employee, origin: String): ObjectNode = fields(employee).apply {
        putArray("links").apply {
            addObject().put("rel", "self").put("href", "$origin/employees/${employee.id}")
            addObject().put("rel", "employees").put("href", "$origin/employees")
        }
    }

    /** Spring's templates: a method in lower case, and every field of an employee, or none for a delete. */
    private fun templates(methods: Map<String, String>): ObjectNode = mapper.createObjectNode().apply {
        methods.forEach { (name, method) ->
            putObject(name).apply {
                put("method", method)
                val properties = putArray("properties")
                if (method != "delete") FIELDS.forEach { properties.addObject().put("name", it) }
            }
        }
    }

    companion object {
        const val HOST_HEADER = "X-Sim-Host"
        private const val HAL_FORMS = "application/prs.hal-forms+json"
        private const val HAL = "application/hal+json"
        private val FIELDS = listOf("firstName", "id", "lastName", "role")
    }
}
SpringCloneTrace.kt, the comparison
package com.artie.heart.sims.employees

import com.fasterxml.jackson.databind.ObjectMapper
import com.fasterxml.jackson.dataformat.yaml.YAMLFactory
import com.fasterxml.jackson.dataformat.yaml.YAMLGenerator
import org.http4k.client.JavaHttpClient
import org.http4k.core.Method
import org.http4k.core.Request
import org.http4k.core.Response
import org.junit.jupiter.api.Test
import java.io.File
import java.time.OffsetDateTime
import java.time.ZoneOffset
import java.time.format.DateTimeFormatter

/**
 * Proves the employees simulator, answering as stock Spring (`/_sim/employees/mode`), is a clone of Spring HATEOAS's
 * own affordances example: every request goes to the clone and to the example itself, run untouched, and the run
 * stops at the first answer that differs in status, Content-Type, Location, ETag or body, byte for byte once the
 * example's origin is written as the clone's (both write every link absolute, against the address they were asked
 * at). It writes the trace for the run report, each step's comparison the example's own answer, only when
 * SPRING_CLONE_OUT names a folder, with SPRING_REAL and SPRING_CLONE the two origins.
 */
class SpringCloneTrace {
    @Test
    fun writesTheComparison() {
        val out = System.getenv("SPRING_CLONE_OUT")?.let(::File) ?: return
        val real = System.getenv("SPRING_REAL") ?: "http://localhost:8097"
        val clone = System.getenv("SPRING_CLONE") ?: "http://localhost:8096"
        out.mkdirs()
        File(out, "trace.yaml").writeText(CloneRun(real, clone).trace())
    }
}

private class CloneRun(private val real: String, private val clone: String) {
    private val mapper = ObjectMapper()
    private val yaml = ObjectMapper(YAMLFactory().disable(YAMLGenerator.Feature.WRITE_DOC_START_MARKER))
    private val client = JavaHttpClient()
    private val steps = mutableListOf<Map<String, Any?>>()

    fun trace(): String {
        val reset = client(Request(Method.POST, "$clone/_sim/reset"))
        check(reset.status.successful) { "the clone did not reset" }
        check(client(Request(Method.PUT, "$clone/_sim/employees/mode").body("stock-spring")).status.successful)
        read()
        acts()
        return yaml.writeValueAsString(mapOf("steps" to steps))
    }

    /** What each media type gets, and what a client's validators get: nothing to revalidate with. */
    private fun read() {
        get("The entry, asked for as {accept}", "/employees", HAL_FORMS)
        get("The entry, asked for as {accept}: Spring's own JSON, links in an array, no forms", "/employees", HAL)
        get("The entry, asked for as {accept}: HAL-FORMS", "/employees", ANY)
        get("Employee {id}, as {accept}", "/employees/1", HAL_FORMS, mapOf("id" to 1))
        get("Employee {id}, as {accept}", "/employees/1", HAL, mapOf("id" to 1))
        send(
            "Employee {id} again, with If-None-Match: no ETag was given, so nothing to revalidate, read in full",
            Request(Method.GET, "/employees/1").header("Accept", HAL_FORMS).header("If-None-Match", "\"1\""),
            mapOf("id" to 1),
        )
        get("Employee {id}, who never was", "/employees/$NEVER", HAL_FORMS, mapOf("id" to NEVER), expect = 404)
    }

    /** The templates' acts, sent upper case (Tomcat refuses `post`), each answered 204 with no body. */
    private fun acts() {
        send(
            "Act {relation} on the entry: hire {firstName}, by POST, answered with no body",
            json(Method.POST, "/employees", SAMWISE), mapOf("relation" to "default", "firstName" to "Samwise"), 204,
        )
        get("The entry again: {count} employees", "/employees", HAL_FORMS, mapOf("count" to 3))
        send(
            "Act {relation} on employee {id}: {role}, by PUT, with an If-Match that matches nothing, taken anyway",
            json(Method.PUT, "/employees/1", FRODO_RETIRED).header("If-Match", "\"stale\""),
            mapOf("relation" to "default", "id" to 1, "role" to "ring bearer, retired"), 204,
        )
        get("Employee {id}, as changed", "/employees/1", HAL_FORMS, mapOf("id" to 1))
        send(
            "Act {relation} on employee {id}, by DELETE, answered with no body",
            Request(Method.DELETE, "/employees/1"), mapOf("relation" to "deleteEmployee", "id" to 1), 204,
        )
        get("Employee {id}, gone", "/employees/1", HAL_FORMS, mapOf("id" to 1), expect = 404)
        get("The entry at the end", "/employees", HAL_FORMS)
    }

    private fun get(
        say: String,
        path: String,
        accept: String,
        params: Map<String, Any?> = emptyMap(),
        expect: Int = 200,
    ) = send(say, Request(Method.GET, path).header("Accept", accept), params + ("accept" to accept), expect)

    private fun json(method: Method, path: String, body: String) =
        Request(method, path).header("Content-Type", JSON).header("Accept", HAL_FORMS).body(body)

    /** [request] to the clone and to the example; recorded, and refused here if the two answers differ. */
    private fun send(say: String, request: Request, params: Map<String, Any?>, expect: Int = 200) {
        val at = OffsetDateTime.now(ZoneOffset.UTC).format(DateTimeFormatter.ISO_OFFSET_DATE_TIME)
        val started = System.nanoTime()
        val cloned = client(request.uri(request.uri.scheme("http").authority(authority(clone))))
        val ms = (System.nanoTime() - started) / NANOS_PER_MILLI
        val original = client(request.uri(request.uri.scheme("http").authority(authority(real))))
        check(cloned.status.code == expect) { "$say: the clone answered ${cloned.status.code}, expected $expect" }
        same(say, cloned, original)
        steps += linkedMapOf(
            "at" to at,
            "mode" to "script",
            "say" to say,
            "params" to params,
            "request" to requestOf(request, clone),
            "response" to responseOf(cloned, clone),
            "compare" to mapOf(
                "label" to "Spring HATEOAS's affordances example, commit c100c56, at $real",
                "diff" to true,
                "request" to requestOf(request, real),
                "response" to responseOf(original, real),
            ),
            "summary" to "${verdictOf(cloned)}, as Spring's example answers, byte for byte",
            "expect" to expect,
            "ms" to ms,
        ) + listOfNotNull(
            // The addresses the run names itself, not judged as hypermedia: the entry, and an employee who never was.
            if (steps.isEmpty() || params["id"] == NEVER) "entry" to true else null,
        )
    }

    /** Every part a client reads, the same: each origin written alike, the bytes otherwise as sent. */
    private fun same(say: String, cloned: Response, original: Response) {
        fun part(response: Response, origin: String) = listOf(
            response.status.code.toString(),
            response.header("Content-Type").orEmpty(),
            response.header("Location").orEmpty().replace(origin, ORIGIN),
            response.header("ETag").orEmpty(),
            response.bodyString().replace(origin, ORIGIN),
        )
        val (a, b) = part(cloned, clone) to part(original, real)
        PARTS.forEachIndexed { i, name ->
            check(a[i] == b[i]) { "$say: $name differs:\n clone: ${a[i]}\n Spring: ${b[i]}" }
        }
    }

    private fun verdictOf(response: Response) = when {
        response.status.code == NO_CONTENT -> "No content"
        response.status.code == NOT_FOUND -> "Not found, with no body"
        else -> "Read"
    }

    private fun authority(origin: String) = origin.removePrefix("http://")

    private fun requestOf(request: Request, origin: String): Map<String, Any?> = linkedMapOf(
        "method" to request.method.name,
        "url" to origin + request.uri.toString(),
        "headers" to listOf("Accept", "Content-Type", "If-Match", "If-None-Match")
            .mapNotNull { name -> request.header(name)?.let { name to it } }.toMap(),
        "body" to request.bodyString().takeIf { it.isNotEmpty() }?.let { mapper.readValue(it, Any::class.java) },
    )

    private fun responseOf(response: Response, origin: String): Map<String, Any?> = linkedMapOf(
        "status" to response.status.code,
        "headers" to listOf("Content-Type", "Location", "ETag", "Content-Length")
            .mapNotNull { name -> response.header(name)?.let { name to it } }.toMap(),
        "body" to response.bodyString().takeIf { it.isNotEmpty() }?.let { mapper.readValue(it, Any::class.java) },
