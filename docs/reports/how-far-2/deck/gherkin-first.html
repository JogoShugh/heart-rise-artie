core/gherkin-first.sh (PreToolUse)
#!/usr/bin/env bash
# PreToolUse (Edit|Write|MultiEdit). Blocks a write to production code unless
# a feature file already has an in-flight change, or the author switched
# refactor mode on (scripts/refactor-mode). Exit 2 blocks the tool call
# and sends stderr back to Claude.
set -o pipefail
. "$(dirname "$0")/lib.sh"

file_path=$(bdd_hook_file_path)
[ -n "$file_path" ] || exit 0
bdd_has_config || exit 0
bdd_load

rel=$(bdd_rel "$file_path")
bdd_is_prod "$rel" || exit 0
bdd_features_dirty && exit 0
# The author switched refactor mode on: pre-commit checks its own terms.
bdd_refactor_on && exit 0

cat >&2 <<EOF
Blocked: about to modify production code ($rel) with no feature file change in this changeset yet.
Write or update the Gherkin scenario first, run the BDD suite to see it fail, then implement.
Feature patterns: $(bdd_list featurePatterns | tr '\n' ' ')
EOF
exit 2

# core/log-event.sh (PostToolUse)
if bdd_is_prod "$rel"; then
  bdd_append "MAIN_WRITE $rel"
elif bdd_is_feature "$rel"; then
  bdd_append "FEATURE_WRITE $rel"
fi

core/log-event.sh (PostToolUse)
#!/usr/bin/env bash
# PostToolUse (Edit|Write|MultiEdit). Records MAIN_WRITE / FEATURE_WRITE facts
# in .claude/tdd-events.log for the pre-commit ordering check. Never blocks.
set -o pipefail
. "$(dirname "$0")/lib.sh"

file_path=$(bdd_hook_file_path)
[ -n "$file_path" ] || exit 0
bdd_has_config || exit 0
bdd_load

rel=$(bdd_rel "$file_path")
if bdd_is_prod "$rel"; then
  bdd_append "MAIN_WRITE $rel"
elif bdd_is_feature "$rel"; then
  bdd_append "FEATURE_WRITE $rel"
fi
exit 0

# core/bdd-run.sh: every run, pass or fail
if [ $ec -eq 0 ]; then
  bdd_append "BDD_RUN PASS exit=0"
else
  bdd_append "BDD_RUN FAIL exit=$ec"
fi

core/bdd-run.sh, every run, pass or fail
#!/usr/bin/env bash
# Runs the BDD suite and records "BDD_RUN PASS|FAIL" in the event log, pass or
# fail. For stacks with bddLogging "wrapper" (no native build hook that runs
# after a failed test task). Extra args are appended to bddCommand.
# Exits with the suite's own exit code.
set -o pipefail
. "$(dirname "$0")/lib.sh"

bdd_has_config || { echo "bdd-run: no $(bdd_config)" >&2; exit 1; }
cmd=$(bdd_value bddCommand)
[ -n "$cmd" ] || { echo "bdd-run: bddCommand not set in $(bdd_config)" >&2; exit 1; }

cd "$(bdd_root)" || exit 1
if [ $# -gt 0 ]; then
  bash -c "$cmd $*"
else
  bash -c "$cmd"
fi
ec=$?

if [ $ec -eq 0 ]; then
  bdd_append "BDD_RUN PASS exit=0"
else
  bdd_append "BDD_RUN FAIL exit=$ec"
fi
exit $ec
