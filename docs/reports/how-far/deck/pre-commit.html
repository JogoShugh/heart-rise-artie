core/pre-commit, the ordering check
feature=$(first_line ' FEATURE_WRITE ')
fail=$(first_line "${run_re}FAIL")
main=$(first_line ' MAIN_WRITE ')
last_run=$(printf '%s\n' "$recent" | grep -E -- "$run_re" | tail -1)

reasons=()
[ -n "$feature" ] || reasons+=("no FEATURE_WRITE since the last commit")
[ -n "$fail" ] || reasons+=("no failing BDD run since the last commit (the scenario was never seen red)")
if [ -n "$feature" ] && [ -n "$main" ] && [ "$feature" -gt "$main" ]; then
  reasons+=("production code was written before any feature file")
fi
if [ -n "$fail" ] && [ -n "$main" ] && [ "$fail" -gt "$main" ]; then
  reasons+=("production code was written before the BDD suite was seen failing")
fi
case "$last_run" in
  *" PASS"*) ;;
  *) reasons+=("the latest BDD run since the last commit is not PASS") ;;
esac

if [ ${#reasons[@]} -gt 0 ]; then
  echo "pre-commit: TDD ordering check FAILED:" >&2
  printf '  - %s\n' "${reasons[@]}" >&2
  echo "" >&2
  echo "Events since the last commit:" >&2
  printf '%s\n' "$recent" | sed 's/^/  /' >&2
  exit 1
fi

echo "pre-commit: TDD ordering check passed."

Three facts, in order

FEATURE_WRITE ↓ BDD_RUN FAIL ↓ MAIN_WRITE … ↓ BDD_RUN PASS

Works for a plain terminal git commit too. --no-verify and git commit -n are blocked for the model by protect-gate.sh.
