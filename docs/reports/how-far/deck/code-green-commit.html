// Detours.kt: keep a read another step names
private fun named(capture: Capture): Set<Int> {
  val marks = capture.steps.associateWith {
    capture.marks(it) }
  return marks.filter { (step, own) ->
    val name = own["as"]?.toString()
      ?: return@filter false
    val mention = Regex("""\{$name[.}]""")
    marks.any { (other, its) -> other.line !=
      step.line && its.values.any {
        mention.containsMatchIn("$it") } }
  }.keys.map { it.number }.toSet()
