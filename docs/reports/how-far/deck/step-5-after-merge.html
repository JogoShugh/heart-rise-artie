Where the reasons come from (github-scm/pull.reader.yaml)
# A pull request says nothing about what is allowed, so the rules are written here:
# first by its state, then by the rules below, in order, each with its reason.
matches:
  path: /repos/{owner}/{repo}/pulls/{pull_number}
allowed:
  table: /state
  rows:
    open:
      allow: [close, merge, edit, request-review, update-branch, comment, label]
      absent: { reopen: The pull request is already open. }
    closed:
      allow: [reopen, edit, comment, label]
      absent:
        close: The pull request is already closed.
        merge: "A closed pull request can't be merged; reopen it first."
        request-review: "A closed pull request can't have reviews requested; reopen it first."
        update-branch: "A closed pull request's branch isn't updated; reopen it first."
  rules:
    # The repository's own answer says whether the caller may push (the fact named in pull.resource.yaml).
    - when: { "repository:/permissions/push": false }
      absent:
        close: You can't push to this repository.
        reopen: You can't push to this repository.
        merge: You can't push to this repository.
        edit: You can't push to this repository.
        request-review: You can't push to this repository.
        update-branch: You can't push to this repository.
    # Labels are triage's, as on an issue.
    - when: { "repository:/permissions/triage": false }
      absent: { label: You need triage access to this repository. }
    # A locked conversation takes comments only from collaborators with push access, as an issue's does.
    - when: { /locked: true, "repository:/permissions/push": false }
      absent: { comment: The conversation is locked; only collaborators can comment. }
    - when: { /merged: true }
      absent: { reopen: A merged pull request can't be reopened. }
    - when: { /draft: true }
      absent: { merge: "A draft can't be merged; mark it ready for review first." }
    # GitHub works mergeability out in the background; until it has, mergeable is null.
    - when: { /mergeable: null }
      absent: { merge: "GitHub is still checking whether it can be merged; read it again shortly." }
    # GitHub says behind only when the base's protection asks for up-to-date branches and names a required
    # check; with none required, a branch behind its base is clean (seen live, 2026-10-16).
    - when: { /mergeable_state: behind }
      absent: { merge: "The branch is behind its base; update it first." }
    # Without that protection, the gateway's own rule: GitHub's compare says the base has commits the branch lacks.
    - when: { /state: open, "compare:/behind_by": { not: 0 } }
      absent: { merge: "The branch is behind its base; update it first." }
    - when: { /mergeable_state: dirty }
      absent:
        merge: "The branch has conflicts with its base; resolve them first."
        update-branch: "The branch has conflicts with its base; resolve them first."
    - when: { /mergeable_state: blocked }
      absent: { merge: "Branch protection blocks the merge: a required check or review is missing." }
